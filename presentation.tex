% ===== SECTION: INTRODUCTION =====
    \section{Introduction}

    \begin{frame}{Context and Motivation}
        \begin{itemize}
            \item Digital labor market: internship listings are abundant but \textbf{semi-structured, inconsistent, and noisy}.
            \item Sources: LinkedIn, Indeed, university portals, direct JSON uploads, each with its own schema.
            \item \textbf{Two extreme approaches fail:}
            \begin{itemize}
                \item \textit{Lexical only (e.g., BM25):} cannot handle synonyms, semantic gap (e.g., “Code Wizard” vs. “Software Engineer”).
                \item \textit{Pure LLM pipeline:} high cost, high latency, privacy concerns, and inconsistency.
            \end{itemize}
            \item \textbf{Goal:} A lightweight, tiered semantic enrichment pipeline that runs on commodity hardware.
        \end{itemize}
    \end{frame}

    \begin{frame}{Problem Statement and Research Questions}
        \textbf{Problem:} Unstructured internship data + semantic gap + heterogeneous JSON inputs.
        \vspace{0.5cm}

        \textbf{Research Questions:}
        \begin{enumerate}
            \item \textbf{RQ1:} How to balance extraction accuracy and computational cost using a tiered pipeline?
            \item \textbf{RQ2:} How well can lightweight transformers map non‑standard titles to the ESCO taxonomy?
            \item \textbf{RQ3:} How much does hybrid retrieval (BM25 + HNSW) improve relevance over keyword‑only search?
            \item \textbf{RQ4:} Can heuristic‑based mapping automate ingestion of arbitrarily structured JSON files?
        \end{enumerate}
    \end{frame}

    \section{System Design}

% --- FIXED TIERED PIPELINE SLIDE (3-column layout, correct LaTeX) ---
    \begin{frame}{Tiered Semantic Enrichment Pipeline}
        \textbf{Design principle:} Cheap, fast methods handle most cases; expensive ones only for edge cases.
        \vspace{0.5cm}

        \begin{columns}[T]
            \begin{column}{0.32\textwidth}
                \centering
                \textcolor{myteal}{\faCheckCircle~ \textbf{Layer A}} \\
                \textcolor{myteal}{\textbf{Deterministic}} \\
                \textcolor{myteal}{\textbf{(78\% of cases)}}
                \vspace{0.2cm}

                \begin{itemize}
                    \item Regex salary parsing (98\% precision)
                    \item Location/date cleaning
                    \item Gazetteer‑based skill extraction (800+ ESCO skills)
                    \item Keyword‑anchored role categorisation
                \end{itemize}
            \end{column}

            \begin{column}{0.32\textwidth}
                \centering
                \textcolor{myaccent}{\faBrain~ \textbf{Layer B}} \\
                \textcolor{myaccent}{\textbf{Lightweight Neural}} \\
                \textcolor{myaccent}{\textbf{(19\% of cases)}}
                \vspace{0.2cm}

                \begin{itemize}
                    \item DistilBERT for ambiguous title disambiguation
                    \item SBERT maps titles to ESCO Level 4
\item Top‑3 accuracy: \textbf{91\%}
\end{itemize}
\end{column}

\begin{column}{0.32\textwidth}
\centering
\textcolor{red}{\faRobot~ \textbf{Layer C}} \\
\textcolor{red}{\textbf{LLM Fallback}} \\
\textcolor{red}{\textbf{(3\% of cases)}}
\vspace{0.2cm}

\begin{itemize}
\item Reserved for highly idiosyncratic titles
\item Example: “Prompt Engineer”
\item GPT‑3.5 / local LLM on demand
\end{itemize}
\end{column}
\end{columns}

\vspace{0.5cm}
\centering
\textcolor{myteal}{\faLongArrowAltRight}~ \textbf{Fallback flow:} Layer A → (if fails) → Layer B → (if fails) → Layer C
\end{frame}

\begin{frame}{Data Ingestion and Schema Normalisation}
\begin{itemize}
\item \textbf{Schema‑on‑read:} fuzzy key matching for JSON uploads (“link”, “url”, “apply\_href” → unified \texttt{sourceUrl}).
\item \textbf{Deterministic ID generation:} \texttt{slugify(title) + "-at-" + slugify(company)} → blocking key.
\item \textbf{Validation (Layer 0):} type guards, range checks, required fields; malformed records quarantined.
\item \textbf{Sanitisation:} HTML entity resolution, whitespace collapsing, boilerplate removal.
\item \textbf{Caching:} enriched objects stored in server‑side Map – enrichment done once per data refresh.
\end{itemize}
\end{frame}

\begin{frame}{Hybrid Retrieval with RRF}
\begin{columns}[T]
\begin{column}{0.5\textwidth}
\textbf{\faKeyboard~ Lexical (BM25)}
\begin{itemize}
\item Inverted index over \texttt{rawTitle} (weight 3), \texttt{rawDescription} (1), \texttt{company} (1.5)
\item Exact matches for skills (“Kubernetes”), acronyms, company names
\end{itemize}
\vspace{0.3cm}
\textbf{\faBrain~ Semantic (HNSW)}
\begin{itemize}
\item Pre‑computed 384‑dim SBERT embeddings
\item HNSW graph: $M=16$, \texttt{efConstruction}=200, $\log N$ search time
\end{itemize}
\end{column}
\begin{column}{0.5\textwidth}
\textbf{\faLink~ Reciprocal Rank Fusion}
\[
\text{RRF\_score}(d) = \sum_{r\in\{\text{bm25},\text{vector}\}} \frac{1}{60 + \text{rank}_r(d)}
\]
\begin{itemize}
\item Scale‑invariant, no score normalisation
\item Robust to outliers
\item Encourages consensus between lexical and semantic signals
\end{itemize}
\end{column}
\end{columns}
\end{frame}

% ===== SECTION: EVALUATION =====
\section{Evaluation}

\begin{frame}{Experimental Setup}
\begin{itemize}
\item \textbf{Dataset:} 327 real internship listings (LinkedIn, Indeed, university portals, direct uploads).
\item \textbf{Ground truth:} 500 manually annotated listings (ESCO Level 4, skills, salary, location), inter‑annotator agreement $\kappa = 0.84$.
\item \textbf{Query set:} 30 queries – 10 lexical, 10 conceptual, 10 mixed.
\item \textbf{Metrics:} Retrieval (P@5, HR@10, NDCG@10, MAP, MRR), Extraction (Precision, Recall, F1, Top‑k), Efficiency (throughput, latency, memory, cost)
\item \textbf{Hardware:} Intel Core i7‑13700K, 32 GB DDR5, no GPU (commodity, ≈ \$1200).
\end{itemize}
\end{frame}

\begin{frame}{Results – Extraction and Classification (RQ1, RQ2)}
\begin{table}
\centering
\begin{tabular}{lccc}
\toprule
Task & Precision & Recall & F1 / Top‑k \\
\midrule
Salary extraction (regex) & 0.98 & 0.81 & – \\
Location extraction & 0.89 & 0.76 & – \\
Skill extraction (gazetteer) & 0.94 & 0.64 & F1=0.71 \\
ESCO title norm. (SBERT) & – & – & Top‑1: 0.78, Top‑3: 0.91 \\
Domain categorisation & – & – & Exact match: 0.86 (heuristic), 0.89 (+embedding) \\
\bottomrule
\end{tabular}
\end{table}
\begin{itemize}
\item Layer A handles 78\% of extractions independently.
\item SBERT improves Top‑1 accuracy by \textbf{24\%} over keyword baseline.
\item Main failure modes: implicit information, relative dates, monthly salary rates.
\end{itemize}
\end{frame}

\end{frame}

\begin{frame}{Error Analysis and Trade‑offs}
\textbf{Extraction error categories:}
\begin{enumerate}
\item Implicit/latent information (48\% of errors) – requires knowledge graph reasoning.
\item Format variance beyond regex coverage (32\%) – e.g., “€3k‑3.5k/mo”.
\item Ambiguity requiring external knowledge (20\%) – e.g., “top‑tier university”.
\end{enumerate}
\vspace{0.3cm}
\textbf{Accuracy–cost Pareto frontier:}
\begin{itemize}
\item Layer A: low cost, moderate recall.
\item Layer B (SBERT) gives large accuracy gains for modest cost.
\item LLM gives +2–3\% absolute improvement at 30–120× higher cost – not justified for bulk processing.
\end{itemize}
\end{frame}

% ===== SECTION: CONCLUSION =====
\section{Conclusion}

\begin{frame}{Contributions}
\begin{enumerate}
\item \textbf{Tiered semantic enrichment pipeline} (deterministic + lightweight neural + bounded LLM fallback).
\item \textbf{Hybrid search infrastructure} (BM25 + HNSW + RRF) using in-memory indices.
\item \textbf{Automated JSON ingestion heuristics} (fuzzy key matching, content‑based inference).
\item \textbf{Empirical evaluation} on 327 real listings, showing:
\begin{itemize}
\item 29\% NDCG@10 improvement over BM25.
\item 47 ms median latency, 6.2 min full ingestion on commodity hardware.
\item 78\% of extractions handled by deterministic rules alone.
\end{itemize}
\end{enumerate}
\end{frame}

\begin{frame}{Future Work}
\begin{itemize}
\item \textbf{Multilingual support:} Cross‑lingual dense retrieval for ESCO.
\item \textbf{Contextual skill weighting:} Section segmentation using BiLSTM‑CRF.
\item \textbf{Self‑training / distillation:} Use LLM to generate rationales and distil into DistilBERT.
\item \textbf{Taxonomy drift monitoring:} Detect emerging skills via OOV n‑gram frequency.
\item \textbf{User interaction integration:} A/B testing, click‑through models, collaborative filtering.
\item \textbf{Comparative evaluation} against LinkedIn Jobs, Indeed, Handshake.
\end{itemize}
\end{frame}

\begin{frame}{Ethical and Practical Considerations}
\begin{itemize}
\item \textbf{Privacy:} Listings may contain PII; production must implement redaction and comply with GDPR/CCPA.
\item \textbf{Bias:} Automated pipelines can inherit societal biases; EU AI Act classifies recruitment systems as “high‑risk”.
\item \textbf{Transparency:} Rule‑based Layer A is fully explainable; embedding layer less so, but attention visualisation helps.
\item \textbf{Deployment:} Tiered design allows incremental investment – start with rules, add neural only where needed. No GPU required → accessible to universities and SMEs.
\end{itemize}
\end{frame}
